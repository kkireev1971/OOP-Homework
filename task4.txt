class Student:
    def __init__(self, first_name, second_name, gender):
        self.first_name = first_name
        self.second_name = second_name
        self.gender = gender
        self.finish_courses = []
        self.courses_in_progress = []
        self.grades = {}


    def add_courses(self, course_name): # Метод добавление курса
        self.finish_courses.append(course_name)
    def __str__(self):
        return f'Имя: {self.first_name}\nФамилия: {self.second_name}\nСредняя оценка за домашние задания: {self.av_student_grade()}\nЗавершенные курсы: {self.finish_courses}\nКурсы в процессе завершения:{self.courses_in_progress}'

    def rate_lecture(self, lector, course, grade): # grading lectors
        if isinstance(lector, Lecturer) and course in lector.courses_attached and course in self.courses_in_progress:
            if course in lector.grades:
                lector.grades[course] += [grade]
            else:
                lector.grades[course] = [grade]
        else:
            return 'Ошибка'

    def av_student_grade(self): #student's average grade
        total = 0
        count = 0
        for x in self.grades.values():
            total = total + sum(x)   # Прибавляем сумму оценок из списка x
            count = count + len(x)   # Прибавляем количество оценок в списке x
        return total / count         # Делим общую сумму на общее количество

    def __gt__(self, other):
        if self.av_student_grade() > other.av_student_grade():
            return f'Средний балл студента: {self.second_name} больше чем у студента: {other.second_name}'
        else:
            return f'Средний балл студента: {other.second_name} больше чем у студента: {self.second_name}'

    def __eq__(self, other):
        if self.av_student_grade() == other.av_student_grade():
            return f'Средний балл студента: {self.second_name} такой же как у студента: {other.second_name}'
        else:
            return f'Средний балл студента: {other.second_name} не равен среднему баллу студента: {self.second_name}'


class Mentor:
    def __init__(self, first_name, second_name):
        self.first_name = first_name
        self.second_name = second_name
        self.courses_attached = []
    def __str__(self):
        return f'{self.first_name},{self.second_name},{self.courses_attached}'

class Lecturer(Mentor):
    def __init__(self,first_name, second_name):
        super().__init__(first_name, second_name)
        self.grades = {}
    def __str__(self):
        return f'Имя: {self.first_name}\nФамилия: {self.second_name}\nСредняя оценка за лекции: {self.av_grade()}'

    def av_grade(self): # lector's average grade
        total = 0
        count = 0
        for x in self.grades.values():
            total = total + sum(x)   # Прибавляем сумму оценок из списка x
            count = count + len(x)   # Прибавляем количество оценок в списке x
        return total / count         # Делим общую сумму на общее количество

    def __gt__(self, other):
        if self.av_grade() > other.av_grade():
            return f'Средний балл лектора: {self.second_name} больше чем у лектора: {other.second_name}'
        else:
            return f'Средний балл лектора: {other.second_name} больше чем у лектора: {self.second_name}'

    def __eq__(self, other):
        if self.av_grade() == other.av_grade():
            return f'Средний балл лектора: {self.second_name} такой же как у лектора: {other.second_name}'
        else:
            return f'Средний балл лектора: {self.second_name} не равен среднему баллу лектора: {other.second_name}'

def av_grade_in_course(list, course): # average grade by course
    total = 0
    count = 0
    for x in list:
        if course in x.grades:
            total = total + sum(x.grades[course])
            count = count + len(x.grades[course])

    return total/count
